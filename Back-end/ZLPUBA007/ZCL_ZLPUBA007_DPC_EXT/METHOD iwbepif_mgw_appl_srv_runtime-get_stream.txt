  METHOD /iwbep/if_mgw_appl_srv_runtime~get_stream.
    DATA: ls_lheader  TYPE ihttpnvp,
          lv_filename TYPE string.

    DATA: lt_keys    TYPE /iwbep/t_mgw_tech_pairs,
          ls_key     TYPE /iwbep/s_mgw_tech_pair,
          ls_header  TYPE zspuba032,
          lv_xstring TYPE xstring,
          ls_stream  TYPE ty_s_media_resource.

    lt_keys = io_tech_request_context->get_keys( ).

    CASE iv_entity_set_name.
      WHEN  'PoPdfSet'.
        READ TABLE lt_keys INTO ls_key
          WITH KEY name = 'PO_LIST'.
        IF sy-subrc = 0.
          ls_header-po_list = ls_key-value.
        ENDIF.

        READ TABLE lt_keys INTO ls_key
          WITH KEY name = 'PO_TYPE'.
        IF sy-subrc = 0.
          ls_header-po_type = ls_key-value.
        ENDIF.

        READ TABLE lt_keys INTO ls_key
          WITH KEY name = 'NO_DISPLAY'.
        IF sy-subrc = 0.
          ls_header-no_display = ls_key-value.
        ENDIF.

        "**>>CHG-01: Insert Start
        READ TABLE lt_keys INTO ls_key
          WITH KEY name = 'NO_BANKACC'.
        IF sy-subrc = 0.
          ls_header-no_bankacc = ls_key-value.
        ENDIF.
        "**>>CHG-01: Insert End

        DATA: ls_selscreen TYPE rsparams,
              lt_selscreen TYPE TABLE OF rsparams.

        SPLIT ls_header-po_list AT ':' INTO TABLE DATA(lt_po_list).

        LOOP AT lt_po_list INTO DATA(ls_po_list).
          CLEAR ls_selscreen.
          ls_selscreen-selname = 'S_EBELN'.
          ls_selscreen-sign    = 'I'.
          ls_selscreen-option  = 'EQ'.
          ls_selscreen-low     = ls_po_list.
          APPEND ls_selscreen TO lt_selscreen.
        ENDLOOP.

        IF ls_header-po_type = '1'.
          CLEAR ls_selscreen.
          ls_selscreen-selname = 'R1'.
          ls_selscreen-sign = 'I'.
          ls_selscreen-option = 'EQ'.
          ls_selscreen-low = 'X'.
          APPEND ls_selscreen TO lt_selscreen.

          CLEAR ls_selscreen.
          ls_selscreen-selname = 'R2'.
          ls_selscreen-sign = 'I'.
          ls_selscreen-option = 'EQ'.
          ls_selscreen-low = ''.
          APPEND ls_selscreen TO lt_selscreen.
        ELSEIF ls_header-po_type = '2'.
          CLEAR ls_selscreen.
          ls_selscreen-selname = 'R1'.
          ls_selscreen-sign = 'I'.
          ls_selscreen-option = 'EQ'.
          ls_selscreen-low = ''.
          APPEND ls_selscreen TO lt_selscreen.

          CLEAR ls_selscreen.
          ls_selscreen-selname = 'R2'.
          ls_selscreen-sign = 'I'.
          ls_selscreen-option = 'EQ'.
          ls_selscreen-low = 'X'.
          APPEND ls_selscreen TO lt_selscreen.
        ENDIF.

        CLEAR ls_selscreen.
        ls_selscreen-selname = 'P_POSCB'.
        ls_selscreen-sign = 'I'.
        ls_selscreen-option = 'EQ'.
        ls_selscreen-low = ls_header-no_display.
        APPEND ls_selscreen TO lt_selscreen.

        CLEAR ls_selscreen.
        ls_selscreen-selname = 'P_FIORI'.
        ls_selscreen-sign = 'I'.
        ls_selscreen-option = 'EQ'.
        ls_selscreen-low = 'X'.
        APPEND ls_selscreen TO lt_selscreen.

        "**>>CHG-01: Insert Start
        CLEAR ls_selscreen.
        ls_selscreen-selname = 'P_BANKCB'.
        ls_selscreen-sign = 'I'.
        ls_selscreen-option = 'EQ'.
        ls_selscreen-low = ls_header-no_bankacc.
        APPEND ls_selscreen TO lt_selscreen.
        "**>>CHG-01: Insert End

        SUBMIT zlpufm003 WITH SELECTION-TABLE lt_selscreen AND RETURN.
        IMPORT lv_po_pdf = lv_xstring FROM MEMORY ID 'PO_PDF'.

        ls_stream-value = lv_xstring.
        ls_stream-mime_type = 'application/pdf'.

      WHEN OTHERS.
    ENDCASE.

    "To set file name in respond header
    lv_filename = sy-datum && sy-uzeit && '.pdf'.
    lv_filename = escape( val = lv_filename
                          format = cl_abap_format=>e_url ).
    ls_lheader-name = 'Content-Disposition'.
    ls_lheader-value = |inline; filename="{ lv_filename }"|.
    set_header( is_header = ls_lheader ).

    copy_data_to_ref( EXPORTING is_data = ls_stream
                      CHANGING  cr_data = er_stream ).
  ENDMETHOD.
